// Advances the persistency count for one MA given the current bar.
// Returns updated count, exit level, pending-exit flag, entry signal, exit signal.
updatePersistency(float price, float maValue, int prevCount, float prevExitLevel, bool prevPendingExit) =>
    newCount       = prevCount
    newExitLevel   = prevExitLevel
    newPendingExit = prevPendingExit
    entrySignal    = false
    exitSignal     = false

    if not na(maValue)
        if useDecisiveExit
            // Triggers depend on user's chosen price reference
            posExitTrigger = pendingExitPriceType == "Close" ? close < maValue : low  < maValue
            negExitTrigger = pendingExitPriceType == "Close" ? close > maValue : high > maValue

            if newCount > 0
                // Currently in positive (above-MA) run
                if newPendingExit
                    if low < newExitLevel
                        // Decisive flip to negative
                        newCount       := -1
                        newPendingExit := false
                        newExitLevel   := na
                        exitSignal     := true
                        // Check for immediate pending exit on the other side
                        if negExitTrigger
                            newPendingExit := true
                            newExitLevel   := high
                    else if negExitTrigger
                        newPendingExit := false
                        newExitLevel   := na
                        newCount       += 1
                    else
                        newCount += 1
                else
                    if posExitTrigger
                        newPendingExit := true
                        newExitLevel   := low
                        newCount       += 1
                    else
                        newCount += 1

            else if newCount < 0
                // Currently in negative (below-MA) run
                if newPendingExit
                    if high > newExitLevel
                        // Decisive flip to positive
                        newCount       := 1
                        newPendingExit := false
                        newExitLevel   := na
                        entrySignal    := true
                        if posExitTrigger
                            newPendingExit := true
                            newExitLevel   := low
                    else if posExitTrigger
                        newPendingExit := false
                        newExitLevel   := na
                        newCount       -= 1
                    else
                        newCount -= 1
                else
                    if negExitTrigger
                        newPendingExit := true
                        newExitLevel   := high
                        newCount       -= 1
                    else
                        newCount -= 1

            else
                // Count is zero — wait for a cross
                if ta.crossover(price, maValue)
                    newCount       := 1
                    newExitLevel   := na
                    newPendingExit := false
                    entrySignal    := true
                else if ta.crossunder(price, maValue)
                    newCount       := -1
                    newExitLevel   := na
                    newPendingExit := false
                    exitSignal     := true
        else
            // Non-decisive exit: simple above/below counting
            if ta.crossover(price, maValue)
                newCount       := 1
                newExitLevel   := na
                newPendingExit := false
                entrySignal    := true
            else if ta.crossunder(price, maValue)
                newCount       := -1
                newExitLevel   := na
                newPendingExit := false
                exitSignal     := true
            else
                if price > maValue
                    if newCount >= 1
                        newCount += 1
                    else
                        newCount    := 1
                        entrySignal := true
                else if price < maValue
                    if newCount <= -1
                        newCount -= 1
                    else
                        newCount   := -1
                        exitSignal := true
                else
                    newCount := newCount > 0 ? newCount + 1 : newCount < 0 ? newCount - 1 : 0

    [newCount, newExitLevel, newPendingExit, entrySignal, exitSignal]

    float totalTimeDiff = 0.0
    for i = 1 to 10
        totalTimeDiff += time[i - 1] - time[i]
    avgTimeDiff = totalTimeDiff / 10
    barsPerDay := (24 * 60 * 60 * 1000) / avgTimeDiff

    barsPerDay := timeframe.isdaily ? 1 : timeframe.isweekly ? 5 : timeframe.ismonthly ? 22 : na

calcReturn(float entryPrice, int countValue) =>
    na(entryPrice) ? na : countValue > 0 ? ((close - entryPrice) / entryPrice) * 100 : -((entryPrice - close) / entryPrice) * 100

    ma1Return := calcReturn(ma1EntryPrice, ma1Count)

    ma2Return := calcReturn(ma2EntryPrice, ma2Count)

// TABLE — ROW POPULATION

populateRow(bool show, int count, color maColor, int length, string maType, bool pendingExit, float maxCount, float returnPct, int benchmark, int rowIdx) =>
    if show
        int col = 0

        if showColorLabels
            table.cell(persistencyTable, col, rowIdx, "", text_color = headerFg, bgcolor = maColor, text_size = tableSize)
            col += 1

        // MA label
        table.cell(persistencyTable, col, rowIdx, str.tostring(length) + " " + maType, text_color = textColor, bgcolor = bgColor, text_size = tableSize)
        col += 1

        // Persistency
        color countBg = count > 0 ? upColor : count < 0 ? downColor : bgColor
        table.cell(persistencyTable, col, rowIdx, str.tostring(count), text_color = textColor, bgcolor = monochromeTable ? bgColor : countBg, text_size = tableSize)
        col += 1

        // Pending exit
        if useDecisiveExit and showPendingExit
            color pendingBg = pendingExit ? (count > 0 ? downColor : upColor) : bgColor
            table.cell(persistencyTable, col, rowIdx, pendingExit ? "Yes" : "No", text_color = textColor, bgcolor = monochromeTable ? bgColor : pendingBg, text_size = tableSize)
            col += 1

        // Max persistency in lookback
        if showMaxBars and isDailyOrHigher
            string maxText  = na(maxCount) ? "N/A" : str.tostring(maxCount)
            string bullet   = na(maxCount) ? "" : " ●"
            color  maxColor = maxCount >= benchmark ? color.green : color.red
            table.cell(persistencyTable, col, rowIdx, maxText + bullet, text_color = maxColor, bgcolor = bgColor, text_size = tableSize)
            col += 1

        // Returns
        if showReturns
            string retText = na(returnPct) ? "N/A" : str.tostring(returnPct, "#.##") + "%"
            table.cell(persistencyTable, col, rowIdx, retText, text_color = textColor, bgcolor = bgColor, text_size = tableSize)
            col += 1
